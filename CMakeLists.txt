cmake_minimum_required(VERSION 3.21)
project(one480 VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
set(LOGO_FILE "${CMAKE_SOURCE_DIR}/pngs-jpgs-jpegs/logo.jpg")
if(NOT EXISTS "${LOGO_FILE}")
 message(FATAL_ERROR "Required logo missing: pngs-jpgs-jpegs/logo.jpg. Add your own logo.jpg before building.")
endif()
add_library(one480core STATIC 480/480.c 480/win10.c 480/win11.c 480/win32frfull.c 480/kernel32.c 480/r44.c 480/gui.c)
target_include_directories(one480core PUBLIC ${CMAKE_SOURCE_DIR}/hs ${CMAKE_SOURCE_DIR}/480)
if(MSVC)
 target_compile_options(one480core PRIVATE /W4 /permissive-)
else()
 target_compile_options(one480core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(one480-cli 480/main.c)
target_link_libraries(one480-cli PRIVATE one480core)
add_executable(one480 WIN32 sdksgui/main.cpp sdksgui/mainwindow.cpp sdksgui/480-take.cpp sdksgui/high.cpp sdksgui/w.cpp resources/resources.qrc)
target_link_libraries(one480 PRIVATE Qt6::Widgets one480core)
target_include_directories(one480 PRIVATE sdksgui)
if(WIN32)
 target_sources(one480 PRIVATE resources/app.rc)
endif()
enable_testing()
foreach(t test testsc hpp cpp actions-github ekct virus-system trojan-delete warm-delete config)
 add_executable(one480-${t} test/${t}.c)
 add_test(NAME ${t} COMMAND one480-${t})
endforeach()
install(TARGETS one480 RUNTIME DESTINATION .)
